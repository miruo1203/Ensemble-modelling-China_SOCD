# --- Initialization and Setup ---
rm(list = ls())
set.seed(666)

# Load required libraries
library(caret)
library(kernlab)
library(ranger)
library(gbm)
library(Cubist)
library(dplyr)

# Model evaluation function
fun.acc <- function(observed, predicted) {
  R2 <- 1 - sum((observed - predicted)^2) / sum((observed - mean(observed))^2)
  SEP2 <- mean((observed - predicted)^2)
  SEP <- sqrt(SEP2)
  bias <- mean(predicted) - mean(observed)
  SEP2c <- sum(((predicted - bias - observed)^2) / length(observed))
  SEPc <- sqrt(SEP2c)
  RPD <- sd(observed) / SEP
  IQ <- quantile(observed, 0.75) - quantile(observed, 0.25)
  RPIQ <- IQ / SEP
  mx <- mean(observed)
  my <- mean(predicted)
  s2x <- var(observed)
  s2y <- var(predicted)
  sxy <- mean((observed - mx) * (predicted - my))
  ccc <- 2 * sxy / (s2x + s2y + (mx - my)^2)
  return(data.frame(R2 = R2, CCC = ccc, RMSE = SEP, Bias = bias, RPD = RPD, RPIQ = RPIQ))
}

# --- Load and preprocess data ---
setwd("C:/Users/Cat/Desktop")  # Change to your working directory
data <- read.csv("Cn_BD_1.csv")
data$CV_LD <- as.factor(data$CV_LD)

# One-hot encoding for categorical variable
dummy_vars <- dummyVars("~ CV_LD", data = data)
data_encoded <- predict(dummy_vars, newdata = data)
model_data <- cbind(data_encoded, data[c(2:9, 11, 12, 13, 14, 15)])

# Split into training and validation sets by profile_id
profiles <- split(model_data, data$profile_id)
profile_ids <- names(profiles)
train_ids <- sample(profile_ids, round(length(profile_ids) * 0.75))
row.cal <- which(data$profile_id %in% train_ids)
train_data <- model_data[row.cal, ]
test_data <- model_data[-row.cal, ]

# Define training control
trControl <- trainControl(method = "cv", number = 5, savePredictions = TRUE)

# --- 1. SVM ---
svm_grid <- expand.grid(sigma = 0.1, C = 10)
svm_model <- train(BD ~ ., data = train_data, method = "svmRadial", tuneGrid = svm_grid, trControl = trControl)
svm_pred <- predict(svm_model, newdata = test_data)

# --- 2. Random Forest ---
rf_grid <- expand.grid(mtry = 3:5, min.node.size = 5, splitrule = "variance")
rf_model <- train(BD ~ ., data = train_data, method = "ranger", trControl = trControl, tuneGrid = rf_grid)
rf_pred <- predict(rf_model, newdata = test_data)

# --- 3. GBM ---
gbm_grid <- expand.grid(interaction.depth = c(30, 40), n.trees = 500, shrinkage = 0.05, n.minobsinnode = 10)
gbm_model <- train(BD ~ ., data = train_data, method = "gbm", trControl = trControl, tuneGrid = gbm_grid, verbose = FALSE)
gbm_pred <- predict(gbm_model, newdata = test_data)

# --- 4. Cubist ---
cubist_grid <- expand.grid(committees = c(50, 100), neighbors = 3:5)
cubist_model <- train(BD ~ ., data = train_data, method = "cubist", tuneGrid = cubist_grid, trControl = trControl)
cubist_pred <- predict(cubist_model, newdata = test_data)

# --- 5. Ensemble Model (EM) ---
obs <- test_data$BD
EM_data <- data.frame(
  obs = obs,
  SVM_pred = svm_pred,
  RF_pred = rf_pred,
  GBM_pred = gbm_pred,
  Cubist_pred = cubist_pred
)

em_lm <- lm(obs ~ SVM_pred + RF_pred + GBM_pred + Cubist_pred, data = EM_data)
em_pred <- predict(em_lm, newdata = EM_data)

# --- Model evaluation ---
eval_svm <- fun.acc(obs, svm_pred)
eval_rf <- fun.acc(obs, rf_pred)
eval_gbm <- fun.acc(obs, gbm_pred)
eval_cubist <- fun.acc(obs, cubist_pred)
eval_em <- fun.acc(obs, em_pred)

# --- Print evaluation results ---
cat("\nSVM Performance:\n"); print(eval_svm)
cat("\nRF Performance:\n"); print(eval_rf)
cat("\nGBM Performance:\n"); print(eval_gbm)
cat("\nCubist Performance:\n"); print(eval_cubist)
cat("\nEM (Ensemble Model) Performance:\n"); print(eval_em)

# --- Export predictions ---
results_all <- data.frame(
  obs = obs,
  SVM_pred = svm_pred,
  RF_pred = rf_pred,
  GBM_pred = gbm_pred,
  Cubist_pred = cubist_pred,
  EM_pred = em_pred
)
write.csv(results_all, "All_Model_Predictions.csv", row.names = FALSE)

# --- Output ensemble model coefficients ---
cat("\nEM model coefficients:\n"); print(coef(em_lm))
