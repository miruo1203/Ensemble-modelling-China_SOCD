# Function for model evaluation
fun.acc <- function(observed, predicted) {
  R2 <- 1 - sum((observed - predicted)^2) / sum((observed - mean(observed))^2)
  SEP2 <- mean((observed - predicted)^2)
  SEP <- sqrt(SEP2)
  bias <- mean(predicted) - mean(observed)
  SEP2c <- sum(((predicted - bias - observed)^2) / length(observed))
  SEPc <- sqrt(SEP2c)
  RPD <- sd(observed) / SEP
  IQ <- quantile(observed, 0.75) - quantile(observed, 0.25)
  RPIQ <- IQ / SEP
  mx <- mean(observed)
  my <- mean(predicted)
  s2x <- var(observed)
  s2y <- var(predicted)
  sxy <- mean((observed - mx) * (predicted - my))
  ccc <- 2 * sxy / (s2x + s2y + (mx - my)^2)
  gf <- data.frame(R2 = R2, concordance = ccc, RMSE = SEP, bias = bias, RPD = RPD, RPIQ = RPIQ, row.names = NULL)
  return(gf)
}
